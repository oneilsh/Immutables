Benchmarking immutables Collections

Method

This article measures individual operation times for each collection type provided by immutables. Each timing cell builds a fresh data structure of size n, then uses microbenchmark to time one operation. We report the median of 10 repetitions per cell.

Comparisons use idiomatic base R implementations as controls:

  • Sequences and queues: base R lists (plus rstackdeque for queues).
  • Priority queues: parallel value and priority vectors with which.min()/which.max().
  • Interval queries: a data.frame with start/end columns, filtered with vectorized comparisons. When the Bioconductor package IRanges is available, it is included as a third comparison.

All immutables structures are persistent — operations return new structures without modifying the input. The base R baselines are copy-on-write but not persistent in the same sense; we benchmark the cost of producing the updated copy.

Sequence operations

Seven operations on sequences of 100 to 500,000 elements. Append, prepend, concatenate, and split exploit the finger tree’s O(log n) structural sharing; get, replace, and remove at the middle exercise index-based splitting.

sizes <- c(100, 500, 1000, 5000, 10000, 50000, 500000)
results <- list()

for(n in sizes) {
  values <- as.list(sprintf("v_%06d", seq_len(n)))
  fs <- as_flexseq(values)
  bl <- values
  mid <- as.integer(n / 2)
  fs2 <- as_flexseq(values)
  bl2 <- values

  # flexseq operations
  gc(FALSE)
  bm <- microbenchmark(push_back(fs, "z"), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "flexseq", op = "append", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(push_front(fs, "z"), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "flexseq", op = "prepend", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(fs[[mid]], times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "flexseq", op = "get middle", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark({ out <- fs; out[[mid]] <- "y"; out }, times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "flexseq", op = "replace middle", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(pop_at(fs, mid)$remaining, times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "flexseq", op = "remove middle", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(c(fs, fs2), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "flexseq", op = "concatenate", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(split_at(fs, mid), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "flexseq", op = "split at middle", n = n, time_us = bm$time / 1000)

  # base R list operations
  gc(FALSE)
  bm <- microbenchmark(c(bl, list("z")), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R list", op = "append", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(c(list("z"), bl), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R list", op = "prepend", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(bl[[mid]], times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R list", op = "get middle", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark({ out <- bl; out[[mid]] <- "y"; out }, times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R list", op = "replace middle", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(bl[-mid], times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R list", op = "remove middle", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(c(bl, bl2), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R list", op = "concatenate", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(list(left = bl[seq_len(mid - 1L)], value = bl[[mid]], right = bl[(mid + 1L):n]), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R list", op = "split at middle", n = n, time_us = bm$time / 1000)
}

       color = "Implementation") +
  theme_bw() + theme(plot.title = element_text(hjust = 0.5), legend.position = "bottom")

Queue operations

FIFO enqueue (push to back) and dequeue (pop from front) compared across flexseq, rstackdeque, and base R lists. Both flexseq and rstackdeque provide O(log n) or amortized O(1) queue operations; base R’s c() and [-1] are O(n).

sizes <- c(100, 500, 1000, 5000, 10000, 50000, 100000)
results <- list()

for(n in sizes) {
  items <- as.list(rep("queue_item", n))
  fs <- as_flexseq(items)
  rq <- rstackdeque::as.rpqueue(items)
  bl <- items

  # flexseq
  gc(FALSE)
  bm <- microbenchmark(push_back(fs, "d"), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "flexseq", op = "enqueue", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(pop_front(fs)$remaining, times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "flexseq", op = "dequeue", n = n, time_us = bm$time / 1000)

  # rstackdeque
  gc(FALSE)
  bm <- microbenchmark(rstackdeque::insert_back(rq, "d"), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "rstackdeque", op = "enqueue", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(rstackdeque::without_front(rq), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "rstackdeque", op = "dequeue", n = n, time_us = bm$time / 1000)

  # base R list
  gc(FALSE)
  bm <- microbenchmark(c(bl, list("d")), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R list", op = "enqueue", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(bl[-1L], times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R list", op = "dequeue", n = n, time_us = bm$time / 1000)
}

queue_results <- do.call(rbind, results)
queue_summary <- aggregate(time_us ~ impl + op + n, data = queue_results, FUN = median)
queue_summary$time_ms <- queue_summary$time_us / 1000

ggplot(queue_summary, aes(x = n, y = time_ms, color = impl)) +
  geom_line(linewidth = 0.6) + geom_point(size = 1.6) +
  facet_wrap(~ op, scales = "free_y") +
  scale_x_log10(labels = scales::label_comma()) +
  scale_y_log10(labels = scales::label_comma()) +
  labs(title = "Queue Operations", x = "Number of elements", y = "Time (ms)",
       color = "Implementation") +
  theme_bw() + theme(plot.title = element_text(hjust = 0.5), legend.position = "bottom")

Priority queue operations

Insert, peek, and pop operations for min and max priority. The base R baseline stores values and priorities as parallel vectors and uses which.min() / which.max() for queries — O(n) per call. The priority_queue from immutables maintains a sorted finger tree with cached min/max monoids, giving O(log n) insert and pop.

sizes <- c(100, 500, 1000, 5000, 10000, 50000)
results <- list()

set.seed(42)
max_n <- max(sizes)
all_values <- sprintf("val_%06d", seq_len(max_n))
all_priorities <- runif(max_n)

for(n in sizes) {
  vals <- as.list(all_values[seq_len(n)])
  pris <- all_priorities[seq_len(n)]
  pq <- as_priority_queue(vals, priorities = pris)
  base_pq <- list(values = all_values[seq_len(n)], priorities = pris)

  # priority_queue
  gc(FALSE)
  bm <- microbenchmark(insert(pq, "val_new", 0.5), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "priority_queue", op = "insert", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(peek_min(pq), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "priority_queue", op = "peek min", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(pop_min(pq)$remaining, times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "priority_queue", op = "pop min", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(peek_max(pq), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "priority_queue", op = "peek max", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(pop_max(pq)$remaining, times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "priority_queue", op = "pop max", n = n, time_us = bm$time / 1000)

  # base R (parallel vectors)
  gc(FALSE)
  bm <- microbenchmark(list(values = c(base_pq$values, "val_new"), priorities = c(base_pq$priorities, 0.5)), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R", op = "insert", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(base_pq$values[which.min(base_pq$priorities)], times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R", op = "peek min", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark({ idx <- which.min(base_pq$priorities); list(values = base_pq$values[-idx], priorities = base_pq$priorities[-idx]) }, times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R", op = "pop min", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(base_pq$values[which.max(base_pq$priorities)], times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R", op = "peek max", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark({ idx <- which.max(base_pq$priorities); list(values = base_pq$values[-idx], priorities = base_pq$priorities[-idx]) }, times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R", op = "pop max", n = n, time_us = bm$time / 1000)
}

pq_results <- do.call(rbind, results)
pq_summary <- aggregate(time_us ~ impl + op + n, data = pq_results, FUN = median)
pq_summary$time_ms <- pq_summary$time_us / 1000

ggplot(pq_summary, aes(x = n, y = time_ms, color = impl)) +
  geom_line(linewidth = 0.6) + geom_point(size = 1.6) +
  facet_wrap(~ op, scales = "free_y") +
  scale_x_log10(labels = scales::label_comma()) +
  scale_y_log10(labels = scales::label_comma()) +
  labs(title = "Priority Queue Operations", x = "Number of elements", y = "Time (ms)",
       color = "Implementation") +
  theme_bw() + theme(plot.title = element_text(hjust = 0.5), legend.position = "bottom")

Interval queries

Insert and three query types (single-point lookup, all-point matches, and range overlap) on a collection of intervals with integer endpoints. The base R baseline stores intervals in a data.frame and filters with vectorized comparisons — simple and fast for small n, but O(n) per query. interval_index uses an augmented finger tree for O(log n + k) queries, where k is the number of matches.

sizes <- c(100, 500, 1000, 5000, 10000, 50000)
results <- list()

set.seed(123)
max_n <- max(sizes)
all_starts <- sort(sample.int(max_n * 10L, max_n))
all_widths <- sample.int(100L, max_n, replace = TRUE)
all_ends <- all_starts + all_widths
all_values <- sprintf("interval_%06d", seq_len(max_n))

query_point <- all_starts[as.integer(max_n / 2)] + 10L
query_lo <- all_starts[as.integer(max_n * 0.4)]
query_hi <- all_starts[as.integer(max_n * 0.5)]

has_iranges <- requireNamespace("IRanges", quietly = TRUE) &&
  requireNamespace("S4Vectors", quietly = TRUE)

for(n in sizes) {
  s <- all_starts[seq_len(n)]
  e <- all_ends[seq_len(n)]
  v <- all_values[seq_len(n)]

  ix <- as_interval_index(as.list(v), start = s, end = e, bounds = "[]")
  df <- data.frame(start = s, end = e, value = v, stringsAsFactors = FALSE)

  ins_start <- all_starts[as.integer(max_n / 2)]
  ins_end <- ins_start + 50L

  # interval_index
  gc(FALSE)
  bm <- microbenchmark(insert(ix, "interval_new", ins_start, ins_end), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "interval_index", op = "insert", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(peek_point(ix, query_point, bounds = "[]"), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "interval_index", op = "point query", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(peek_all_point(ix, query_point, bounds = "[]"), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "interval_index", op = "all point matches", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(peek_all_overlaps(ix, query_lo, query_hi, bounds = "[]"), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "interval_index", op = "overlap query", n = n, time_us = bm$time / 1000)

  # base R data.frame
  gc(FALSE)
  bm <- microbenchmark(rbind(df, data.frame(start = ins_start, end = ins_end, value = "interval_new", stringsAsFactors = FALSE)), times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R", op = "insert", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark({ hit <- which(df$start <= query_point & query_point <= df$end); if(length(hit)) df$value[hit[1L]] else NULL }, times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R", op = "point query", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(df[df$start <= query_point & query_point <= df$end, , drop = FALSE], times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R", op = "all point matches", n = n, time_us = bm$time / 1000)

  gc(FALSE)
  bm <- microbenchmark(df[df$start <= query_hi & df$end >= query_lo, , drop = FALSE], times = repeats)
  results[[length(results) + 1]] <- data.frame(impl = "base R", op = "overlap query", n = n, time_us = bm$time / 1000)

  # IRanges (if available)
  if(has_iranges) {
    ir <- list(iranges = IRanges::IRanges(start = s, end = e), values = v)

    gc(FALSE)
    bm <- microbenchmark(list(iranges = c(ir$iranges, IRanges::IRanges(start = ins_start, end = ins_end)), values = c(ir$values, "interval_new")), times = repeats)
    results[[length(results) + 1]] <- data.frame(impl = "IRanges", op = "insert", n = n, time_us = bm$time / 1000)

    gc(FALSE)
    bm <- microbenchmark({ q <- IRanges::IRanges(start = query_point, width = 1L); hits <- IRanges::findOverlaps(q, ir$iranges); sh <- S4Vectors::subjectHits(hits); if(length(sh)) ir$values[sh[1L]] else NULL }, times = repeats)
    results[[length(results) + 1]] <- data.frame(impl = "IRanges", op = "point query", n = n, time_us = bm$time / 1000)

    gc(FALSE)
    bm <- microbenchmark({ q <- IRanges::IRanges(start = query_point, width = 1L); hits <- IRanges::findOverlaps(q, ir$iranges); ir$values[S4Vectors::subjectHits(hits)] }, times = repeats)
    results[[length(results) + 1]] <- data.frame(impl = "IRanges", op = "all point matches", n = n, time_us = bm$time / 1000)

    gc(FALSE)
    bm <- microbenchmark({ q <- IRanges::IRanges(start = query_lo, end = query_hi); hits <- IRanges::findOverlaps(q, ir$iranges); ir$values[S4Vectors::subjectHits(hits)] }, times = repeats)
    results[[length(results) + 1]] <- data.frame(impl = "IRanges", op = "overlap query", n = n, time_us = bm$time / 1000)
  }
}

ivx_results <- do.call(rbind, results)
ivx_summary <- aggregate(time_us ~ impl + op + n, data = ivx_results, FUN = median)
ivx_summary$time_ms <- ivx_summary$time_us / 1000

ggplot(ivx_summary, aes(x = n, y = time_ms, color = impl)) +
  geom_line(linewidth = 0.6) + geom_point(size = 1.6) +
  facet_wrap(~ op, scales = "free_y") +
  scale_x_log10(labels = scales::label_comma()) +
  scale_y_log10(labels = scales::label_comma()) +
  labs(title = "Interval Index Queries", x = "Number of elements", y = "Time (ms)",
       color = "Implementation") +
  theme_bw() + theme(plot.title = element_text(hjust = 0.5), legend.position = "bottom")

Notes

  • All immutables operations are persistent: the original structure is never modified. Base R list operations produce shallow copies via copy-on-write.
  • Sequence append and prepend are O(log n) for flexseq vs O(n) for base R; the crossover is visible at moderate sizes.
  • Get middle ([[) is O(log n) for flexseq vs O(1) for base R — flexseq will always be slower here. This is the cost of tree-based indexing.
  • Queue and priority queue operations show the expected O(log n) vs O(n) separation at larger sizes.
  • Interval query performance depends on the number of matches (k). The benchmark uses a fixed query point and range, so k varies with n and the random interval layout.
  • These figures are meant for article exposition rather than internal regression tracking.
